function defaults() {
  return {
    name:"Dreamer",
    onboarded:false,
    theme:"Pink",
    preferences:{
      trackers:[],
      features:[],
      layout:"Cosy and organised",
      reminderPreference:"Only a few reminders",
      checkin:"Yes please"
    },
    trackers:[],
    entries:[],
    tasks:[],

    journal:[],
    notes:[],

    reminders:[],
    checkins:[]
  };
}

function loadData() {
  try {
    const saved = localStorage.getItem(STORAGE_KEY);
    if(saved) {
      const parsed = JSON.parse(saved);
      data = {...defaults(),...parsed};
      data.preferences = {
        ...defaults().preferences,
        ...(parsed.preferences||{})
      };
    } else {
      data = defaults();
    }
  } catch(e) {
    data = defaults();
  }
}

function save() {
  try {
    localStorage.setItem(STORAGE_KEY,JSON.stringify(data));
  } catch(e) {
    alert("Your browser could not save the changes. Check your storage settings.");
  }
}

function $(id) {
  return document.getElementById(id);
}

function safe(value) {
  return String(value ?? "").replace(/[&<>"']/g,c=>({
    "&":"&amp;",
    "<":"&lt;",
    ">":"&gt;",
    '"':"&quot;",
    "'":"&#39;"
  })[c]);
}

function uid() {
  return Date.now().toString(36)+Math.random().toString(36).slice(2,8);
}

function today() {
  const d = new Date();
  return [
    d.getFullYear(),
    String(d.getMonth()+1).padStart(2,"0"),
    String(d.getDate()).padStart(2,"0")
  ].join("-");
}

function dateText() {
  return new Date().toLocaleDateString("en-US",{
    weekday:"long",
    month:"long",
    day:"numeric",
    year:"numeric"
  });
}

function applyTheme() {
  const t = themes[data.theme] || themes.Pink;
  const root = document.documentElement;

  ["--bg","--card","--text","--accent","--soft","--border"].forEach((key,i)=>{
    root.style.setProperty(key,t[i]);
  });
}

function startApp() {
  if(data.onboarded) {
    openDashboard();
    return;
  }

  $("welcome").classList.add("hidden");
  $("survey").classList.remove("hidden");

  qIndex=0;
  answers={};
  renderQuestion();
}

function renderQuestion() {
  const q=questions[qIndex];

  $("surveyProgress").style.width =
    ((qIndex+1)/questions.length*100)+"%";

  let html=`<h2>${safe(q.title)}</h2>`;

  if(q.type==="text") {
    html+=`
      <input id="answerText"
        placeholder="${safe(q.placeholder)}"
        value="${safe(answers.name||"")}">
    `;
  } else {
    html+=`<div class="option-grid">`;

    q.options.forEach(o=>{
      const selected=q.type==="multi"
        ?(answers[q.key]||[]).includes(o)
        :answers[q.key]===o;

      html+=`
        <button class="option ${selected?"selected":""}"
          onclick="pickAnswer('${safe(o)}')">
          ${safe(o)}
        </button>
      `;
    });

    html+="</div>";
  }

  $("surveyQuestion").innerHTML=html;
}

function pickAnswer(value) {
  const q=questions[qIndex];

  if(q.type==="multi") {
    let arr=answers[q.key]||[];

    answers[q.key]=arr.includes(value)
      ?arr.filter(x=>x!==value)
      :[...arr,value];
  } else {
    answers[q.key]=value;
  }

  renderQuestion();
}

function surveyNext() {
  const q=questions[qIndex];

  if(q.type==="text") {
    answers.name=$("answerText").value.trim()||"Dreamer";
  }

  if(q.type==="multi"&&!answers[q.key]) {
    answers[q.key]=[];
  }

  if(q.type==="single"&&!answers[q.key]) {
    alert("Choose an option to continue.");
    return;
  }

  if(qIndex<questions.length-1) {
    qIndex++;
    renderQuestion();
  } else {
    finishSurvey();
  }
}

function surveyBack() {
  if(qIndex>0) {
    qIndex--;
    renderQuestion();
  } else {
    $("survey").classList.add("hidden");
    $("welcome").classList.remove("hidden");
  }
}

function finishSurvey() {
  data.name=answers.name||"Dreamer";

  data.preferences={
    trackers:answers.trackers||[],
    features:answers.features||[],
    layout:answers.layout||"Cosy and organised",
    reminderPreference:answers.reminderPreference||"Only a few reminders",
    checkin:answers.checkin||"Yes please"
  };

  data.theme=answers.theme||"Pink";
  data.onboarded=true;

  data.trackers=data.preferences.trackers.map(name=>({
    id:uid(),
    name,
    unit:
      name==="Reading"?"pages":
      name==="Water"?"glasses":
      name==="Money savings"?"AED":
      name==="Sleep"?"hours":"times",
    target:1
  }));

  save();
  openDashboard();
}

function buildNav() {
  const nav=$("nav");
  nav.innerHTML="";

  pageList.forEach(page=>{
    const feature=pageFeature[page];

    if(feature && !(data.preferences.features||[]).includes(feature)) return;

    const button=document.createElement("button");
    button.textContent=page;
    button.id="nav"+page;
    button.onclick=()=>showPage(page);
    nav.appendChild(button);
  });
}

function showPage(page) {
  currentPage=page;

  document.querySelectorAll(".nav button").forEach(b=>b.classList.remove("active"));

  if($("nav"+page)) $("nav"+page).classList.add("active");

  const renderers={
    "Home":homePage,
    "Trackers":trackersPage,
    "To-do":tasksPage,
    "Goals":goalsPage,
    "Calendar":calendarPage,
    "Habits":habitsPage,
    "Progress":progressPage,

    "Journal":journalPage,
    "Notes":notesPage,
    "Countdowns":countdownsPage,
    "Reminders":remindersPage,

}

    </div>

    <div id="trackerHistory"></div>

function addTracker() {
  const name=$("trackerName").value.trim();
  const unit=$("trackerUnit").value.trim()||"times";

  if(!name) return alert("Give your tracker a name.");

  if(data.trackers.some(t=>t.name.toLowerCase()===name.toLowerCase())) {
    return alert("That tracker already exists.");
  }

  data.trackers.push({
    id:uid(),name,unit,
    target:Number($("trackerTarget").value)||0
  });

  save();
  trackersPage();
}

function removeTracker(id) {
  if(!confirm("Delete this tracker and its entries?")) return;

  data.trackers=data.trackers.filter(t=>t.id!==id);
  data.entries=data.entries.filter(e=>e.trackerId!==id);

  save();
  trackersPage();
}

function toggleFeature(feature,checked) {
  const list=data.preferences.features;

  data.preferences.features=checked
    ?[...new Set([...list,feature])]
    :list.filter(f=>f!==feature);

  save();
  buildNav();
}

function savePreferences() {
  data.preferences.layout=$("settingsLayout").value;
  data.preferences.reminderPreference=$("settingsReminder").value;

  save();
  alert("Your preferences have been saved! ♡");
  settingsPage();
}

function exportData() {
  const blob=new Blob(
    [JSON.stringify(data,null,2)],
    {type:"application/json"}
  );

  const url=URL.createObjectURL(blob);
  const a=document.createElement("a");

  a.href=url;
  a.download="daydream-backup.json";
  a.click();

  URL.revokeObjectURL(url);
}

function resetApp() {
  if(!confirm("Are you sure? This will permanently delete your saved DAYDREAM data from this browser.")) return;

  localStorage.removeItem(STORAGE_KEY);
  data=defaults();

  $("dashboard").classList.add("hidden");
  $("survey").classList.add("hidden");
  $("welcome").classList.remove("hidden");
}
